import copy
import random
import unittest
from .read_catalog import ReadCatalog


class CatalogTests(unittest.TestCase):
    def compare(self, actual, expected):
        self.assertEqual(actual, expected)
        for key in [('book', str(n)) for n in range(12)] + [None, ['mutable']]:
            self.assertEqual(key in actual, key in expected)
            for bounds in [(), (1,), (-5, 99), (0, 0)]:
                values = []
                for seq in [actual, expected]:
                    try: values.append(('value', seq.index(key, *bounds)))
                    except (ValueError, TypeError) as error: values.append(type(error))
                self.assertEqual(*values)

    def test_duplicate_order_and_mutations_match_original_lists(self):
        rng = random.Random(411)
        actual = ReadCatalog(); expected = []
        for _ in range(500):
            key = ('book', str(rng.randrange(12)))
            operation = rng.randrange(10)
            rng_count = rng.randrange(2)
            for seq in [actual, expected]:
                if operation == 0: seq.append(key)
                elif operation == 1: seq.extend([key, key])
                elif operation == 2: seq.insert(0, key)
                elif operation == 3 and seq: seq.pop(0)
                elif operation == 4: seq[1:3] = [key]
                elif operation == 5: del seq[2:4]
                elif operation == 6: seq.reverse()
                elif operation == 7: seq.sort()
                elif operation == 8: seq *= rng_count
                elif operation == 9: seq += [key]
            self.compare(actual, expected)

    def test_failed_and_reentrant_iterable_mutation_invalidates_index(self):
        for method in ['extend', '__iadd__']:
            seq = ReadCatalog([('book', '0')]); self.assertEqual(seq.index(('book', '0')), 0)
            def values():
                yield ('book', '1')
                self.assertEqual(seq.index(('book', '1')), 1)
                yield ('book', '2')
                self.assertEqual(seq.index(('book', '2')), 2)
                raise RuntimeError('original iterable failure')
            with self.assertRaisesRegex(RuntimeError, 'original iterable'):
                getattr(seq, method)(values())
            self.compare(seq, [('book', '0'), ('book', '1'), ('book', '2')])

    def test_unhashable_and_mutable_values_keep_original_equality(self):
        value = ['original']; seq = ReadCatalog([value, ('book', '1')])
        self.assertEqual(seq.index(['original']), 0)
        value[0] = 'changed'
        self.assertIn(['changed'], seq)
        self.compare(seq, [value, ('book', '1')])

    def test_remaining_mutators_copies_and_missing_values(self):
        seq = ReadCatalog([('book', '1'), ('book', '1'), ('book', '2')])
        self.assertEqual(seq.index(('book', '1')), 0)
        seq.remove(('book', '1')); self.compare(seq, [('book', '1'), ('book', '2')])
        seq[0] = ('book', '3'); self.compare(seq, [('book', '3'), ('book', '2')])
        for duplicate in [copy.copy(seq), copy.deepcopy(seq)]:
            duplicate.append(('book', '4')); self.assertNotIn(('book', '4'), seq)
            self.compare(duplicate, [('book', '3'), ('book', '2'), ('book', '4')])
        seq.clear(); self.compare(seq, [])


if __name__ == '__main__': unittest.main()
